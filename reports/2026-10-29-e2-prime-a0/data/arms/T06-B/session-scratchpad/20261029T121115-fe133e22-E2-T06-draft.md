def _lead_check_is_dict(lead: dict[str, Any], contract: dict[str, Any] | None,
                        now: datetime | None) -> str | None:
    if not isinstance(lead, dict):
        return "release lead is not an object"
    return None


def _lead_check_qualification(lead: dict[str, Any], contract: dict[str, Any] | None,
                              now: datetime | None) -> str | None:
    if lead.get("qualification") != "verified_release":
        return "bundled release evidence is pending or not verified"
    return None


def _lead_check_required_strings(lead: dict[str, Any], contract: dict[str, Any] | None,
                                 now: datetime | None) -> str | None:
    for key in ("id", "provider", "host_id", "model", "model_version",
                "model_fingerprint", "role"):
        if not isinstance(lead.get(key), str) or not str(lead[key]).strip():
            return f"release lead is missing {key}"
    return None


def _lead_check_provider(lead: dict[str, Any], contract: dict[str, Any] | None,
                         now: datetime | None) -> str | None:
    if str(lead.get("provider")) not in ("openai", "anthropic"):
        return "release lead provider is not a supported hosted lead"
    return None


def _lead_check_role(lead: dict[str, Any], contract: dict[str, Any] | None,
                     now: datetime | None) -> str | None:
    if str(lead.get("role")) != "lead":
        return "release lead must be qualified for the lead role"
    return None


def _lead_check_model_fingerprint(lead: dict[str, Any], contract: dict[str, Any] | None,
                                  now: datetime | None) -> str | None:
    expected = digest({"provider": lead["provider"], "model": lead["model"],
                       "model_version": lead["model_version"]})
    if not hmac.compare_digest(str(lead["model_fingerprint"]), expected):
        return "release lead model fingerprint does not match its exact model version"
    return None


def _lead_check_endpoint_dict(lead: dict[str, Any], contract: dict[str, Any] | None,
                              now: datetime | None) -> str | None:
    endpoint = lead.get("endpoint")
    if not isinstance(endpoint, dict) or not str(endpoint.get("protocol") or "").strip() or \
            not str(endpoint.get("base_url") or "").strip() or \
            not str(endpoint.get("auth_ref") or "").strip():
        return "release lead has no exact endpoint requirement"
    return None


def _lead_check_endpoint_protocol(lead: dict[str, Any], contract: dict[str, Any] | None,
                                  now: datetime | None) -> str | None:
    endpoint = lead["endpoint"]
    if str(endpoint.get("protocol")) != str(lead.get("provider")):
        return "release lead endpoint protocol does not match its provider"
    return None


def _lead_check_endpoint_fingerprint(lead: dict[str, Any], contract: dict[str, Any] | None,
                                     now: datetime | None) -> str | None:
    if not isinstance(lead.get("endpoint_fingerprint"), str) or \
            not lead["endpoint_fingerprint"].strip():
        return "release lead has no endpoint fingerprint"
    if not hmac.compare_digest(str(lead["endpoint_fingerprint"]),
                               endpoint_fingerprint(lead["endpoint"])):
        return "release lead endpoint fingerprint does not match its endpoint requirement"
    return None


def _lead_check_contract_versions(lead: dict[str, Any], contract: dict[str, Any] | None,
                                  now: datetime | None) -> str | None:
    for key in ("protocol_version", "tool_contract_version", "prompt_contract_version"):
        if not isinstance(lead.get(key), str) or not lead[key].strip():
            return f"release lead has no {key.replace('_', ' ')}"
    return None


def _lead_check_core_version(lead: dict[str, Any], contract: dict[str, Any] | None,
                             now: datetime | None) -> str | None:
    if not (isinstance(lead.get("core_version"), str) and lead["core_version"].strip()) and \
            not (isinstance(lead.get("core_version_range"), str) and
                 lead["core_version_range"].strip()):
        return "release lead has no Core version compatibility"
    return None


def _lead_check_dates(lead: dict[str, Any], contract: dict[str, Any] | None,
                      now: datetime | None) -> str | None:
    tested = _as_time(lead.get("tested_at"))
    expiry = _as_time(lead.get("expires_at"))
    current_time = now or datetime.now(timezone.utc)
    if tested is None:
        return "release lead has no valid evidence date"
    if tested > current_time + timedelta(days=1):
        return "release lead evidence date is in the future"
    if expiry is None:
        return "release lead has no valid expiry"
    if expiry <= current_time:
        return "bundled release evidence has expired"
    if expiry <= tested:
        return "release lead expiry must be later than its evidence date"
    return None


def _lead_check_l3_runs(lead: dict[str, Any], contract: dict[str, Any] | None,
                        now: datetime | None) -> str | None:
    try:
        runs = int(lead.get("l3_runs") or 0)
    except (TypeError, ValueError):
        runs = 0
    if runs < L3_REQUIRED_RUNS:
        return f"release lead records {runs} L3 runs, not {L3_REQUIRED_RUNS}"
    return None


def _lead_check_artifacts(lead: dict[str, Any], contract: dict[str, Any] | None,
                          now: datetime | None) -> str | None:
    artifacts = lead.get("qualification_artifacts")
    if not isinstance(artifacts, list) or not artifacts:
        return "release lead has no L3 qualification artifacts"
    for item in artifacts:
        if not isinstance(item, dict) or not any(
                isinstance(item.get(key), str) and item[key].strip()
                for key in ("uri", "path", "id")):
            return "release lead has an invalid qualification artifact reference"
        artifact_hash = item.get("sha256")
        if not isinstance(artifact_hash, str) or \
                not re.fullmatch(r"[0-9a-fA-F]{64}", artifact_hash):
            return "release lead qualification artifact has no SHA-256 integrity record"
    return None


def _lead_check_contract_compat(lead: dict[str, Any], contract: dict[str, Any] | None,
                                now: datetime | None) -> str | None:
    if contract is None:
        return None
    expected_protocol = str(contract.get("protocol_version") or "")
    if expected_protocol and lead["protocol_version"] != expected_protocol:
        return "release lead Core protocol is incompatible"
    expected_tool = str(contract.get("tool_contract_version") or "")
    if expected_tool and lead["tool_contract_version"] != expected_tool:
        return "release lead tool contract is incompatible"
    expected_prompt = str(contract.get("prompt_contract_version") or "")
    if expected_prompt and lead["prompt_contract_version"] != expected_prompt:
        return "release lead prompt contract is incompatible"
    expected_core = str(contract.get("core_version") or "")
    if expected_core:
        exact = str(lead.get("core_version") or "")
        version_range = str(lead.get("core_version_range") or "")
        if exact and exact != expected_core:
            return "release lead Core version is incompatible"
        if not exact and not _in_version_range(expected_core, version_range):
            return "release lead Core version range is incompatible"
    expected_fingerprint = str(contract.get("endpoint_fingerprint") or "")
    if expected_fingerprint and lead["endpoint_fingerprint"] != expected_fingerprint:
        return "release lead endpoint fingerprint changed"
    expected_endpoint = contract.get("endpoint")
    if isinstance(expected_endpoint, dict):
        endpoint = lead["endpoint"]
        if str(endpoint.get("protocol") or "") != str(expected_endpoint.get("protocol") or "") or \
                str(endpoint.get("base_url") or "").rstrip("/") != \
                str(expected_endpoint.get("base_url") or "").rstrip("/") or \
                str(endpoint.get("auth_ref") or "") != str(expected_endpoint.get("auth_ref") or ""):
            return "release lead endpoint requirement changed"
    return None


_RELEASE_LEAD_CHECKS = (
    _lead_check_is_dict,
    _lead_check_qualification,
    _lead_check_required_strings,
    _lead_check_provider,
    _lead_check_role,
    _lead_check_model_fingerprint,
    _lead_check_endpoint_dict,
    _lead_check_endpoint_protocol,
    _lead_check_endpoint_fingerprint,
    _lead_check_contract_versions,
    _lead_check_core_version,
    _lead_check_dates,
    _lead_check_l3_runs,
    _lead_check_artifacts,
    _lead_check_contract_compat,
)


def validate_release_lead(lead: dict[str, Any], *, contract: dict[str, Any] | None = None,
                          now: datetime | None = None) -> tuple[bool, str]:
    """Validate one evidence-bearing release lead, failing closed by detail.

    A pending candidate is deliberately useful to first-run UI, but cannot
    pass this function.  A verified row needs explicit version, endpoint,
    L3-artifact and expiry facts before the Core will combine it with a
    user's L1 probe.
    """
    for check in _RELEASE_LEAD_CHECKS:
        reason = check(lead, contract, now)
        if reason:
            return False, reason
    return True, ""
